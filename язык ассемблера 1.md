-----------------------------------------------------------------------------------------------------
movq <scr>, <dst> -
mov - копирование регистра 
q - 2 байт или 64 битные значения
<scr> - источник откуда
<dst> - приемщик куда
Примеры 
movq %RAX, %PBX          # % ЭТО ОБРАЩЕНИЕ К РЕГИСТРУ
movq (%RAX), %PBX        # () ОПЕРАЦИЯ РАЗЫМЕНОВАНИЕ 
movq $42, %PBX           # ПЕРЕД КОНКРЕТНЫМ ЗНАЧЕНИЕМ ЗНАК ДОЛЛАРА
movq 42, %PBX            # 42 БУДЕТ АДРЕСОМ 
movq $vallue,  %PBX      # vallue БУДЕТ АДРЕСОМ
movq vallue,  %PBX       # ОПЕРАЦИЯ РАЗЫМЕНОВАНИЕ 
-----------------------------------------------------------------------
addq <scr>, <dst> -СЛОЖЕНИЕ,СУММА СОХРАНЯЕТСЯ ВО ВТОРОЕ ЗНАЧЕНИЕ

addq %RAX, %PBX             # ДОБАВЛЕНИЕ ПЕРВОГО КО ВТОРОМУ 
addq %PBX, vallue           # БЕРЕТ ЗНАЧЕНИЕ И ДОБАВЛЯЕТ К ЗНАЧЕНИЮ ПО АДРЕСУ vallue 
movq $42, %PBX              # ДОБАВЛЕНИЕ ОПРЕДЕЛЕННОГО ЗНАЧЕНИЯ 

subq <scr>, <dst> - ВЫЧИТАНИЕ, РАЗНИЦА ХРАНИТСЯ ВО ВТОРОМ ЗНАЧЕНИИ
-----------------------------------------------------------------------------------
 incq<op> - увеличение на единицу
 decq<op> - уменьшение на единицу

 mulq <op> - УМНОЖЕНИЕ
   RAX = (<op>* RAX) ВСЕГДА RAX (МЕНЬШЕ 64 БИТ В RAX)
   RDX = (<op>* RAX) ВСЕГДА RAX (БОЛЬШЕ 64 БИТ В RDX)

 divq <op> -ДЕЛЕНИЕ 
    RAX = (RDX * <op> + RAX) ВСЕГДА RAX (МЕНЬШЕ 64 БИТ В RAX) целое
    RDX = (RDX * <op> + RAX) ВСЕГДА RAX (БОЛЬЩН 64 БИТ В RAX) ОСТАТОК
